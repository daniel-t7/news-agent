```python
# feedparser разбира RSS/Atom XML и го превръща в Python обекти.
import feedparser
# httpx изтегля страници по HTTP (по-модерна алтернатива на requests).
import httpx


# url: адресът на фийда; limit: колко статии да върне (по подразбиране 10).
# Връща списък със статии, а при грешка речник {"error": ...}, затова "list | dict".
def fetch_feed(url: str, limit: int = 10) -> list[dict] | dict:
    # Docstring: документация, достъпна през fetch_feed.__doc__.
    """Връща последните статии от RSS фийд: заглавие, линк, дата."""
    # try/except хваща мрежови грешки, за да не се срине цялата програма.
    try:
        # GET заявка; резултатът r е Response обект (статус, заглавки, тяло).
        r = httpx.get(
            # Адресът, който теглим.
            url,
            # Ако сървърът не отговори за 15 секунди, хвърля грешка вместо да виси.
            timeout=15,
            # Следва пренасочвания (301/302), които са честа практика при фийдове.
            follow_redirects=True,
            # Представяме се със собствено име; някои сайтове отказват заявки без него.
            headers={"User-Agent": "news-agent/0.1"},
        )
        # Ако статусът е 4xx/5xx, хвърля изключение. Статус 200 с грешно
        # съдържание НЕ се хваща тук (точно това ти се случи с първия фийд).
        r.raise_for_status()
    # httpx.HTTPError покрива таймаути, липса на връзка и лоши статуси.
    except httpx.HTTPError as e:
        # Връщаме грешката като данни, за да я види моделът и да реши какво да прави.
        return {"error": f"Неуспешно изтегляне на фийда: {e}"}

    # r.text е тялото като текст; feedparser го разчита и пълни feed.entries.
    # Той не хвърля грешки при лошо съдържание, просто дава празен списък.
    feed = feedparser.parse(r.text)
    # Списъчен израз: за всяка статия строи речник само с трите полета, които ни трябват.
    return [
        {
            # .get(ключ, "") връща "", ако полето липсва, вместо да хвърли KeyError.
            "title": entry.get("title", ""),
            # Линкът към статията.
            "link": entry.get("link", ""),
            # Датата на публикуване като текст, както я дава фийдът.
            "published": entry.get("published", ""),
        }
        # Взимаме само първите limit статии ([:limit] е срез на списъка).
        for entry in feed.entries[:limit]
    ]

# Описание на инструмента за модела. Моделът не вижда кода ти, само това.
FETCH_FEED_SCHEMA = {
    # Формат, който Ollama очаква: инструментът е от тип "function".
    "type": "function",
    "function": {
        # Името трябва да съвпада с ключа в речника на инструментите по-късно.
        "name": "fetch_feed",
        # От този текст моделът решава КОГАТО и ЗАЩО да го извика.
        "description": "Взема последните статии от RSS фийд. "
                       "Връща списък със заглавие, линк и дата на публикация.",
        # Описание на аргументите във формат JSON Schema.
        "parameters": {
            # Аргументите се подават като обект (речник).
            "type": "object",
            # Всеки аргумент с тип и описание, за да знае моделът какво да подаде.
            "properties": {
                "url": {"type": "string", "description": "URL адрес на RSS фийда"},
                "limit": {"type": "integer", "description": "Максимален брой статии (по подразбиране 10)"},
            },
            # Само url е задължителен; ако моделът пропусне limit, важи 10.
            "required": ["url"],
        },
    },
}


# Този блок се изпълнява само при директно пускане на файла (за тест).
if __name__ == "__main__":
    # Вграденият модул json; импортът е тук, защото трябва само за теста.
    import json
    # Вика функцията с реален фийд и ограничава резултата до 3 статии.
    result = fetch_feed("https://hnrss.org/frontpage", limit=3)
    # indent=2 прави изхода четим; ensure_ascii=False запазва кирилицата
    # като букви, а не като \u041a последователности.
    print(json.dumps(result, indent=2, ensure_ascii=False))
```

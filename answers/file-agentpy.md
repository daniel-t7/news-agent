```python
# Вграденият модул за JSON; трябва ни, за да превърнем резултатите на
# инструментите в текст (messages приема само текст в "content").
import json

# Нашите собствени модули. ask праща заявка към Ollama. SCHEMAS е списъкът
# с описания на инструментите (за модела), run_tool ги изпълнява по име.
from news_agent.llm import ask
from news_agent.tools import SCHEMAS, run_tool

# Предпазител: най-много 10 кръга "модел -> инструмент -> модел", за да не
# зацикли агентът безкрайно, ако моделът никога не даде финален отговор.
MAX_STEPS = 10

# Списък със стойности по подразбиране: откъде агентът да чете новини.
# Втората линия е търсене в Hacker News по ключови думи (AI и LLM).
SOURCES = [
    "https://hnrss.org/frontpage",
    "https://hnrss.org/newest?q=AI+LLM",
]

# Системният промпт: правилата, които моделът следва през целия разговор.
# Тройните кавички позволяват текст на няколко реда.
# Стъпките са нумерирани и "ЗАДЪЛЖИТЕЛНО" е с главни, защото малките
# модели иначе пропускат стъпки (това ти се случи).
SYSTEM_PROMPT = """Ти си агент, който следи новини по зададена тема.
Работи строго по този ред:
1. Вземи статиите от всеки източник с fetch_feed.
2. Избери най-много 3 статии, свързани с темата.
3. ЗАДЪЛЖИТЕЛНО извикай read_article за всяка избрана статия.
4. Пиши резюме САМО по прочетения текст. Никога не пиши резюме само по заглавие.
5. Формат за всяка статия: заглавие, 2-3 изречения резюме, линк към статията.
Не измисляй факти. Ако статия не се чете, пропусни я и го спомени.
Пиши на правилен, естествен български. Запазвай техническите термини
(DoS, LLM, GPU, IOMMU) непреведени. Казвай "отворен код", не "отворени източници".
Преди отговор прегледай текста за правописни грешки и слепени думи."""


# Основната функция: приема тема (текст) и връща готовия дайджест (текст).
def run_agent(topic: str) -> str:
    # Историята на разговора като списък от речници. Започва със системните
    # инструкции и задачата. ", ".join(SOURCES) слепва адресите със запетаи.
    # ": list" казва на Pyright да приема и Message обекти, не само речници.
    messages: list = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": f"Тема: {topic}\nИзточници: " + ", ".join(SOURCES)},
    ]

    # Брояч на успешно прочетените статии; ползва се от защитата по-долу.
    articles_read = 0

    # Цикълът на агента: step е 0, 1, 2 ... до MAX_STEPS-1.
    for step in range(MAX_STEPS):
        # Праща цялата история и схемите на инструментите, получава отговор.
        msg = ask(messages, tools=SCHEMAS)
        # Записваме отговора на модела в историята. Без това при следващия
        # кръг моделът няма да "помни", че е поискал инструмент.
        messages.append(msg)

        # Празен tool_calls значи, че моделът не иска инструмент: отговорът е финален.
        if not msg.tool_calls:
            # Защита: моделът иска да приключи, без да е чел нищо
            # (и не сме на последния шанс). Връщаме го с напомняне.
            if articles_read == 0 and step < MAX_STEPS - 1:
                messages.append({
                    "role": "user",
                    "content": "Не си прочел нито една статия. Извикай read_article "
                               "за избраните статии, преди да напишеш дайджеста.",
                })
                # continue пропуска останалото в цикъла и започва следващия кръг.
                continue
            # Финалният дайджест. "or ''" замества None с празен низ.
            return msg.content or ""

        # Моделът може да поиска няколко инструмента наведнъж; минаваме през всички.
        for call in msg.tool_calls:
            # Името на инструмента, който моделът иска (напр. "fetch_feed").
            name = call.function.name
            # Аргументите, които е избрал (напр. {"url": "...", "limit": 10}).
            args = call.function.arguments
            # Дебъг печат, за да виждаш какво решава моделът.
            print(f"[стъпка {step + 1}] {name}({args})")

            # Изпълнява инструмента. dict(args) преобразува аргументите
            # в обикновен речник (за Pyright и run_tool).
            result = run_tool(name, dict(args))

            # Броим само успешните четения (при грешка резултатът е {"error": ...}).
            if name == "read_article" and "text" in result:
                articles_read += 1

            # Връщаме резултата на модела с роля "tool". tool_name му казва
            # кой инструмент е отговорил; json.dumps превръща речника в текст,
            # ensure_ascii=False запазва кирилицата четима.
            messages.append({
                "role": "tool",
                "tool_name": name,
                "content": json.dumps(result, ensure_ascii=False),
            })

    # Стига се до тук само ако цикълът изчерпи всички стъпки без финален отговор.
    return "Достигнат лимит на стъпките без финален отговор."


# Тест: изпълнява се само при директно пускане (python -m news_agent.agent).
if __name__ == "__main__":
    print(run_agent("изкуствен интелект и големи езикови модели"))
```

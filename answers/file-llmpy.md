```python
# Вкарва функцията chat от библиотеката ollama. Тя праща HTTP заявка към
# локалния сървър на Ollama (localhost:11434) и връща отговора на модела.
from ollama import chat

# Името на модела е в константа (главни букви по конвенция), за да се сменя
# на едно място, а не във всяко извикване.
MODEL = "qwen3:8b"

# messages: списък от речници, всеки с "role" и "content" (цялата история).
# tools: списък със схеми на инструменти; "| None = None" значи, че е по избор
# и без него моделът просто отговаря с текст.
def ask(messages: list[dict], tools:list[dict] | None = None):
    # Docstring: документация на функцията (не е коментар, виж по-горе).
    """Праща съобщенията към модела и връща неговото съобщение
        (с полета .content и .tool_calls)."""
    # Изпраща заявката и чака пълния отговор. Подадено tools=None се третира
    # като "без инструменти". Резултатът е обект с отговора и метаданни.
    response = chat(model=MODEL, messages=messages, tools = tools)
    # От целия отговор връщаме само съобщението на модела: .content е текстът,
    # .tool_calls е списък с инструменти, които моделът иска да извикаш.
    return response.message

# Истина само ако пускаш файла директно (python -m news_agent.llm).
# Когато друг файл го импортира, този блок не се изпълнява.
if __name__ == "__main__":
    # Тестово съобщение: списък с един речник, роля "user" (ти говориш на модела).
    msg = ask([{"role":"user", "content":"Кажи здравей на български."}])
    # Печатаме само текста от отговора на модела.
    print(msg.content)
```
